\begin{abstract}
IRM2 calibrates a latent-state neural stochastic differential equation (SDE)
to the U.S.\ Treasury yield curve and Treasury futures. A recurrent encoder
maps a window of observed curves to a latent initial condition; a Neural SDE
evolves it under the risk-neutral measure $\QQ$; a decoder produces the short
rate; and a Monte-Carlo pricer values zero-coupon bonds and the
cheapest-to-deliver futures, the latter through an auxiliary
deliverable-bond network (\emph{BondNet}). This report specifies the model
construction (\S\ref{sec:model}), the pricing maps (\S\ref{sec:pricing}) and
the $\QQ$-objective (\S\ref{sec:obj}); catalogues the problems found during
development by type --- mathematical, architectural and engineering
(\S\ref{sec:problems}); describes the optional physical-measure extension
(\S\ref{sec:pq}); and lists open questions (\S\ref{sec:future}). It assumes
familiarity with stochastic calculus and is concise on implementation.
\end{abstract}

\section{Model construction}\label{sec:model}

The model is a composition $\;\text{history}\xrightarrow{E_\phi} z_0
\xrightarrow{\text{SDE}_\theta} (z_s)_{s\le T_{\max}}
\xrightarrow{D_\psi} (r_s) \to \text{prices}$, with an auxiliary network
$B_\chi$ for deliverable bonds. We describe each block.

\subsection{Encoder}\label{sec:encoder}
Fix a lookback length $L$ and stride $\nu$. The encoder input at date $t$ is
the window
\begin{equation}
X_t=\big[\,\mathbf c_{t-\nu(L-1)},\dots,\mathbf c_{t-\nu},\mathbf c_{t}\,;\,
x_{t-\nu(L-1)},\dots,x_t\,\big]\in\R^{L\times(M+1)},
\end{equation}
the $M$ SVENY yield pillars stacked with the overnight short rate $x$. A fixed
preprocessing $\pi$ rescales the input to percent units, $\pi(X)=100\,X$: raw
decimal yields move by $\sim 5\times10^{-4}$ day-to-day, too weak a signal for
the recurrent cell to separate days (see E\ref{prob:enc}). The latent initial
state is
\begin{equation}
z_0=\mathcal N\!\Big(W\,\mathrm{LSTM}_\phi^{\leftrightarrow}\!\big(\pi(X_t)\big)_{L}+b\Big)\in\R^{d},
\end{equation}
a bidirectional LSTM (last time step), an affine readout to $\R^d$, and an
output normalisation $\mathcal N\in\{\text{LayerNorm},\text{RMSNorm}\}$. The
normalisation fixes $\lVert z_0\rVert$, so the \emph{direction} of $z_0$
carries the day; its magnitude is standardised. A hierarchical variant
combines a fast and a slow stream (different $\nu$) by concatenation,
projection, or addition. $z_0$ is the \emph{only} day-specific quantity in the
model --- a design choice with consequences (A\ref{prob:z0}).

\subsection{Latent dynamics (Neural SDE)}\label{sec:sde}
The latent state solves, under $\QQ$,
\begin{equation}
\dd z_s=\mu_\theta(s,z_s)\,\dd s+\sigma_\theta(s,z_s)\,\dd W_s^{\QQ},
\qquad z_s\in\R^d,\;\; W^{\QQ}\ \text{a $d$-dim Brownian motion},
\label{eq:sde}
\end{equation}
with diagonal noise. Time enters the networks as an extra input,
$(s,z)\mapsto[z\,|\,s]$, so the dynamics are time-inhomogeneous. Two drift
families:
\begin{equation}
\text{(simple)}\ \ \mu_\theta=f_\theta(s,z),
\qquad
\text{(OU)}\ \ \mu_\theta=\kappa_\theta(s,z)\odot\big(\vartheta_\theta(s,z)-z\big),
\quad 0\le\kappa_\theta\le\kappa_{\max},
\end{equation}
where $f_\theta,\vartheta_\theta,\kappa_\theta$ are MLPs (softplus on
$\kappa_\theta$ for non-negativity; the cap $\kappa_{\max}$ is explained in
M\ref{prob:ou}). The diffusion $\sigma_\theta=\eta\,\zeta_\theta(s,z)$,
$\zeta_\theta\ge 0$, where the scalar $\eta$ sets the magnitude
\emph{by construction} (M\ref{prob:vol}) and $\zeta_\theta$ is a constant or an
MLP. Integration is Euler--Maruyama on a uniform grid
$\{s_k=k\Delta\}$, $\Delta=1/\texttt{spy}$, over $N$ Monte-Carlo paths sharing
$z_0$.

\subsection{Decoder and short-rate anchor}\label{sec:decoder}
A decoder $D_\psi:\R^d\to\R$ maps latent state to short rate, with an additive
anchor pinning the front to the observed overnight rate $x_t$:
\begin{equation}
r_s=D_\psi(z_s)+\big(x_t-D_\psi(z_0)\big)
\;\Longrightarrow\;
r_s=x_t+\big(D_\psi(z_s)-D_\psi(z_0)\big),\quad r_0=x_t.
\label{eq:anchor}
\end{equation}
(The bracket is path-independent since all paths share $z_0$.) With a linear
$D_\psi=w^\top z+b$ the short rate is a scalar projection
$r_s=x_t+w^\top(z_s-z_0)$ --- effectively one factor regardless of $d$
(A\ref{prob:onefactor}); a multilayer $D_\psi$ restores curvature.

\subsection{BondNet: deliverable bond pricing}\label{sec:bondnet}
Pricing futures \eqref{eq:futures} requires, at the delivery date $T_f$, the
forward price of each deliverable bond as a function of the latent state ---
itself a conditional expectation $\E^{\QQ}[\,\cdot\mid z_{T_f}]$. Rather than a
nested simulation, a network $B_\chi$ learns it directly. Each deliverable $i$
carries static features
\begin{equation}
b_i=\big(\text{yrs-to-maturity},\,\text{yrs-to-next-coupon},\,\text{yrs-from-last-coupon},\,
c_i,\,f_i,\,n_i,\,\text{accr.\ frac},\,\text{accr.\ int}\big)\in\R^{8},
\end{equation}
($c_i$ coupon, $f_i$ frequency, $n_i$ remaining coupons). The two-branch
late-fusion form (\texttt{SimpleBondNet}) is
\begin{equation}
B_\chi(z,b_i)=\rho\Big(\,g_{\text{fus}}\big[\,g_z(z)\,;\,g_b(b_i)\,\big]\Big),
\qquad \rho=\text{softplus (positivity)},
\end{equation}
with MLP branches $g_z,g_b$ and a fusion head $g_{\text{fus}}$ returning a
scalar dirty price per $100$ face. The output bias is initialised near par
($\$100$) so training starts in range (E\ref{prob:nan}); a feature-wise
modulated variant (\texttt{FiLMBondNet}) is also available. Crucially $B_\chi$
is, a priori, an \emph{unconstrained} function of $z$ --- it is not tied to the
discount factors of the simulated rates, which is the architectural defect
A\ref{prob:decouple} and motivates the consistency term \eqref{eq:cons}.

\section{Risk-neutral pricing}\label{sec:pricing}

\subsection{Zero-coupon bonds and yields}
The pathwise discount factor uses the left-Riemann integral
$I^{(n)}_\tau=\Delta\sum_{s_k\le\tau} r^{(n)}_{s_k}$, with the maturity mapped
to the grid by $k(\tau)=\mathrm{round}(\tau\cdot\texttt{spy})$. The price and
yield are
\begin{equation}
P(0,\tau)=\E^{\QQ}\!\big[e^{-I_\tau}\big]
\;\overset{\text{MC}}{=}\; \exp\!\Big(\lse_{n}\big(-I^{(n)}_\tau\big)-\log N\Big),
\qquad y(\tau)=-\frac{\log P(0,\tau)}{\tau},
\end{equation}
the $\lse$ form being the numerically stable Monte-Carlo mean (clamped to
$P\in[10^{-12},1]$). Maturities are the integer pillars $\tau\in\{1,\dots,M\}$.

\paragraph{Convexity.} With $r_s=x_t+(D_\psi(z_s)-D_\psi(z_0))$ a second-order
cumulant expansion of $\log\E^{\QQ}[e^{-J}]$, $J=\int_0^\tau D_\psi(z_s)\dd s$,
gives
\begin{equation}
y(\tau)\approx
\underbrace{\big(x_t-D_\psi(z_0)\big)}_{\text{level}}
+\underbrace{\tfrac1\tau\!\int_0^\tau\!\E[D_\psi(z_s)]\dd s}_{\text{expectations}}
-\underbrace{\tfrac1{2\tau}\Var\!\Big(\!\int_0^\tau\! D_\psi(z_s)\dd s\Big)}_{\text{convexity}} .
\label{eq:convexity}
\end{equation}
The convexity scales with the squared implied short-rate vol
$\sigma_r\approx\lVert\nabla D_\psi\rVert\,\sigma_z$: a realistic
$\sigma_r\sim1\%$/yr yields $\sim 20$ bp at $10$y; an uncontrolled
$\sigma_r\sim40\%$/yr yields hundreds of percent, saturating the clamp
(M\ref{prob:vol},\,M\ref{prob:convexity}).

\subsection{Treasury futures (cheapest-to-deliver)}
Futures are daily-settled, hence undiscounted: the price is the $\QQ$-expected
CTD ratio at delivery. Let $\iota_f=\max\{k:s_k\le T_f\}$ be the no-look-ahead
grid index at delivery (a \texttt{searchsorted}); the latent state there is
$z_{\iota_f}$. For a contract with deliverable basket $\mathcal B$ and
conversion factors $\{\cf_i\}$,
\begin{equation}
F=\E^{\QQ}\!\Big[\min_{i\in\mathcal B}\frac{B_\chi(z_{\iota_f},b_i)}{\cf_i}\Big]
\;\overset{\text{MC}}{=}\;
\frac1N\sum_{n}\min_{i\in\mathcal B}\frac{B_\chi(z^{(n)}_{\iota_f},b_i)}{\cf_i}.
\label{eq:futures}
\end{equation}
The inner $\min$ is the cheapest-to-deliver option (the short's choice);
across the several contracts active on a date the per-basket minima are
computed by a single segmented reduction (scatter-\texttt{amin}). Active
contracts are those quoted, not yet delivered, with delivery within the
simulation horizon. The short-rate target $\E[r_0]=x_t$ is exact by the anchor
\eqref{eq:anchor} and carries no information (E\ref{prob:deadsr}).

\section{Risk-neutral training objective}\label{sec:obj}
Under $\QQ$ alone the objective is
\begin{equation}
\mathcal L_{\QQ}=
\lambda_y\mathcal L_{\text{yield}}
+\lambda_f\mathcal L_{\text{fut}}
+\lambda_c\mathcal L_{\text{cons}}
+\lambda_\sigma\mathcal L_{\text{vol}} .
\label{eq:objective}
\end{equation}
\begin{itemize}[leftmargin=1.4em]
\item \textbf{Yield} --- absolute MSE on decimals,
$\mathcal L_{\text{yield}}=\frac1M\sum_\tau(y(\tau)-y^\star(\tau))^2$.
\item \textbf{Futures} --- \emph{relative} error, dimensionless so weights are
commensurable, $\mathcal L_{\text{fut}}=\frac1{|\mathcal F|}\sum_k\big((F_k-F^\star_k)/F^\star_k\big)^2$
(E\ref{prob:scale}).
\item \textbf{BondNet--SDE consistency (LSMC).} Reconstruct each deliverable's
cashflows from $b_i$ and discount them \emph{pathwise} on the model's own
rates, $\hat B_i^{(n)}=\sum_j c_{ij}\,e^{-(I^{(n)}_{t_{ij}}-I^{(n)}_{T_f})}$;
then
\begin{equation}
\mathcal L_{\text{cons}}=\tfrac1{100^2}\,\E_n\!\Big[\big(B_\chi(z^{(n)}_{T_f},b_i)-\sg[\hat B_i^{(n)}]\big)^2\Big],
\label{eq:cons}
\end{equation}
a Longstaff--Schwartz regression onto the model-consistent conditional bond
price, with stop-gradient $\sg[\cdot]$ so the gradient reaches BondNet only
(A\ref{prob:decouple}).
\item \textbf{Volatility anchor.}
$\mathcal L_{\text{vol}}=(\std_n[r^{(n)}_{1\mathrm y}]-\sigma^\star)^2$ pins the
implied vol to a historically measured $\sigma^\star\approx1\%$
(M\ref{prob:vol}).
\end{itemize}

\section{Problems encountered}\label{sec:problems}
The development history is best read as a sequence of \emph{identifiability}
and \emph{conditioning} failures, grouped here by type.

\subsection{Mathematical}
\begin{enumerate}[leftmargin=2em,label=\textbf{M\arabic*.}]
\item\label{prob:vol} \textbf{Volatility is not identified.} Yields depend on
$\sigma$ only through the bp-level convexity in \eqref{eq:convexity}; futures
only weakly via CTD optionality. Left free, $\sigma_r$ trained to
$0.1$--$0.4\%$/yr and the Monte-Carlo fan degenerated to a line.
\emph{Resolution:} the anchor $\mathcal L_{\text{vol}}$, legitimate because by
Girsanov $\sigma$ is measure-invariant.
\item\label{prob:convexity} \textbf{Convexity blow-up.} A diffusion magnitude
left to the softplus floor gave $\sigma_r\sim40\%$/yr; the convexity term then
swamps every yield and the discount factor saturates.
\emph{Resolution:} the multiplicative scale $\eta$ sets $\sigma_r\approx1\%$ by
construction.
\item\label{prob:ou} \textbf{OU erases the encoder signal.} For the OU family,
$\E[r_s]$ relaxes to its long-run mean over $1/\kappa$, giving
$y(\tau)\approx (x_t-D_0)+A\,g(\kappa\tau)$ with
$g(x)=1-(1-e^{-x})/x\to1$ as $\kappa\to\infty$: a large $\kappa$ flattens the
$1$--$10$y curve to a constant, independent of $z_0$.
\emph{Resolution:} cap $\kappa\le\kappa_{\max}$ so $z_0$ survives the horizon.
\end{enumerate}

\subsection{Architectural}
\begin{enumerate}[leftmargin=2em,label=\textbf{A\arabic*.}]
\item\label{prob:z0} \textbf{Single-$z_0$ bottleneck.} All day-specific
information enters through the initial condition; the drift, diffusion and
decoder are global. The model can only re-aim a fixed dynamical system per
day, which both limits expressiveness and makes M\ref{prob:ou} possible.
\item\label{prob:onefactor} \textbf{One-factor decoder.} A linear $D_\psi$
makes $r$ a scalar projection of $z$ (\S\ref{sec:decoder}); the curve is then a
one-factor object regardless of $d$.
\emph{Mitigation:} a multilayer decoder (a grid axis).
\item\label{prob:decouple} \textbf{Decoupled futures head.} BondNet
\eqref{eq:futures} is a free function of $z$; nothing ties it to the discount
factors of the same paths, so the model is not internally arbitrage-consistent
and ``joint calibration'' is two heads on a shared encoder.
\emph{Mitigation:} the LSMC term \eqref{eq:cons}. A non-detached version,
however, let the (dominant) consistency gradient flow into the SDE and bent the
curve toward BondNet's par prices --- a $-250$ to $-300$ bp long-end droop ---
hence the stop-gradient.
\end{enumerate}

\subsection{Engineering / training}
\begin{enumerate}[leftmargin=2em,label=\textbf{E\arabic*.}]
\item\label{prob:nan} \textbf{Gradient explosion / NaN losses.} Back-propagating
through a $\sim\!10^3$-step Euler unroll multiplies one Jacobian per step;
large coefficients made the product diverge.
\emph{Resolution:} shorter step $\Delta=1/32$, near-identity SDE
initialisation, coefficient scales, gradient clipping, a non-finite-gradient
optimiser guard, and BondNet's near-par output init.
\item\label{prob:dropout} \textbf{Stochastic SDE coefficients.} Dropout inside
the drift/diffusion MLPs draws a fresh mask each Euler step, so the
\emph{trained} process differs from the deterministic \emph{evaluated} one (a
measure mismatch) and injects unmodelled noise. \emph{Resolution:} dropout
removed from all SDE networks.
\item\label{prob:enc} \textbf{Encoder input scale.} Raw decimal yields move by
$\sim 5\times10^{-4}$/day; the encoder could not separate days, crippling
$z_0$. \emph{Resolution:} percent-unit preprocessing $\pi(X)=100X$.
\item\label{prob:deadsr} \textbf{Dead short-rate target.} $\E[r_0]=x_t$ by the
anchor, so its MSE is $\approx0$ by construction. \emph{Resolution:} weight $0$.
\item\label{prob:scale} \textbf{Loss-scale imbalance.} Absolute futures MSE on
$\$120$ prices exceeds the decimal yield MSE by $10^3$--$10^5$, so no small
$\lambda_f$ balances them. \emph{Resolution:} relative futures error.
\item\label{prob:select} \textbf{Selection on the wrong metric.} The aggregate
loss is dominated by anchor / variance-floor terms, so it ranks models by
those, not by curve fit. \emph{Recommendation:} select on a per-maturity yield
RMSE (bp) on a held-out temporal fold.
\item\label{prob:data} \textbf{Data and unit faults.} (i) Rates were percent
in CSV but the pricer expects decimals; (ii) a $365$-vs-$252$ day-count
inconsistency between the calendar-year maturity grid and date-derived futures
horizons mis-timed the deliverable state; (iii) a single $\cf=\mathrm{NaN}$
poisoned the CTD $\min$. \emph{Resolution:} per-loader $\div100$, a single
business-day/$252$ convention, and basket sanitisation.
\end{enumerate}

\section{The physical-measure ($\PP/\QQ$) extension}\label{sec:pq}
This section is logically separate: it is an \emph{optional} extension, not
part of the core $\QQ$-calibration, and is off by default.

The drift in \eqref{eq:sde} is the risk-neutral drift $\mu^{\QQ}$. With a
market price of risk $\lambda\in\R^d$ and
$\dd W^{\QQ}=\dd W^{\PP}+\lambda\,\dd s$, Girsanov gives the physical drift
\begin{equation}
\mu^{\PP}(s,z)=\mu^{\QQ}(s,z)+\sigma_\theta(s,z)\,\lambda,
\end{equation}
the diffusion being measure-invariant. A one-step physical forecast of the
short rate is matched to the realised future rate $h$ ahead,
\begin{equation}
\hat r^{\PP}_{t+h}=x_t+D_\psi\big(z_0+\mu^{\PP}(0,z_0)h\big)-D_\psi(z_0),
\qquad
\mathcal L_{\PP/\QQ}=\big(\hat r^{\PP}_{t+h}-x_{t+h}\big)^2,
\end{equation}
added to the objective as $\lambda_{\PP}\mathcal L_{\PP/\QQ}$. This identifies
$\lambda$ --- the term premium, since the instantaneous forward equals the
expected future rate plus $\sigma\lambda$. With $\lambda_{\PP}=0$ the model is a
pure $\QQ$-calibration and $\lambda\equiv0$. The grid runs both as an axis. The
current implementation is deliberately minimal (constant $\lambda$, a local
one-step drift match); a fuller treatment is noted in \S\ref{sec:future}.

\section{Identifiability summary and open problems}\label{sec:future}
\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Quantity & Identified by the data? & Mechanism \\
\midrule
curve level / slope & yes (yields) & $\mathcal L_{\text{yield}}$ \\
short-rate volatility & \textbf{no} (vol-insensitive) & $\mathcal L_{\text{vol}}$ anchor (M\ref{prob:vol}) \\
deliverable bond prices & only via CTD min & $\mathcal L_{\text{cons}}$ (A\ref{prob:decouple}) \\
term premium $\lambda$ & \textbf{no} under $\QQ$ alone & $\mathcal L_{\PP/\QQ}$ (\S\ref{sec:pq}) \\
\bottomrule
\end{tabular}
\end{center}

\noindent\textbf{Open problems.}
\begin{enumerate}[leftmargin=*]
\item \emph{Day-conditioned dynamics.} Feed the encoder embedding into
$\mu_\theta$ (and the OU $\vartheta_\theta$), not just the initial condition;
this attacks A\ref{prob:z0} and M\ref{prob:ou} at the root.
\item \emph{Multi-factor decoder.} Resolve A\ref{prob:onefactor} empirically via
the decoder grid axis.
\item \emph{Honest selection.} Per-maturity bp RMSE on a validation fold
(E\ref{prob:select}).
\item \emph{Richer $\PP$ model.} State-dependent $\lambda(s,z)$ and a
$\PP$-transition-density likelihood over the window, giving a time-varying term
premium.
\item \emph{Arbitrage-consistent futures.} Price deliverables directly off the
simulated curve, so \eqref{eq:futures} is consistent with the yields by
construction rather than via an expectation-matched regression.
\end{enumerate}

\paragraph{Status.} The front of the curve fits to $\pm25$ bp, training is
stable, futures price to $\sim1\%$. The failure mode under investigation is a
long-end drift droop; the detached LSMC term, the volatility anchor and the
day-conditioning above target it, with falsifiable signatures: a
non-degenerate fan of width $\sim\sigma^\star\sqrt\tau$, a flat per-maturity
error, and (with the extension) a recovered $\lambda$ consistent with a
positive term premium.
